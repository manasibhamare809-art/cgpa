#include #include using namespace std; class Customer { public: int id; string name; Customer(int i, string n) { id = i; name = n; } }; class Transaction { public: string type; double amount; Transaction(string t, double a) { type = t; amount = a; } }; class Account { public: int accNo; Customer customer; double balance; vector history; Account(int a, Customer c) : customer(c) { accNo = a; balance = 0; } void deposit(double a) { if (a <= 0) return; balance += a; history.push_back(Transaction("Deposit", a)); cout << "Deposit Successful!\n"; } void withdraw(double a) { if (a <= 0) cout << "Invalid Amount!\n"; else if (a > balance) cout << "Insufficient Balance!\n"; else { balance -= a; history.push_back(Transaction("Withdraw", a)); cout << "Withdrawal Successful!\n"; } } void transfer(Account &r, double a) { if (a <= 0 || a > balance || accNo == r.accNo) cout << "Transfer Failed!\n"; else { balance -= a; r.balance += a; history.push_back(Transaction("Transfer", a)); r.history.push_back(Transaction("Received", a)); cout << "Transfer Successful!\n"; } } void display() { cout << "\nAccount: " << accNo; cout << "\nCustomer: " << customer.name; cout << "\nBalance: Rs." << balance << endl; } void showHistory() { cout << "\nTransaction History:\n"; if (history.empty()) cout << "No transactions\n"; for (auto t : history) cout << t.type << " Rs." << t.amount << endl; } }; int main() { vector customers; vector accounts; int choice, id = 1, acc = 101, a, b; double amount; string name; do { cout << "\n--- BANKING SYSTEM ---"; cout << "\n1.Create Customer\n2.Create Account"; cout << "\n3.Deposit\n4.Withdraw\n5.Transfer"; cout << "\n6.Account Info\n7.Transaction History"; cout << "\n8.Exit\nEnter choice: "; cin >> choice; if (choice == 1) { cout << "Enter name: "; cin >> name; customers.emplace_back(id++, name); cout << "Customer Created! ID: " << id - 1 << endl; } else if (choice == 2) { cout << "Enter Customer ID: "; cin >> a; if (a > 0 && a <= customers.size()) { accounts.emplace_back(acc++, customers[a-1]); cout << "Account Created! Number: " << acc-1 << endl; } else cout << "Customer Not Found!\n"; } else if (choice >= 3 && choice <= 7) { cout << "Enter Account Number: "; cin >> a; int i = -1, j = -1; for (int k = 0; k < accounts.size(); k++) if (accounts[k].accNo == a) i = k; if (i == -1) { cout << "Account Not Found!\n"; continue; } if (choice == 3 || choice == 4) { cout << "Enter amount: "; cin >> amount; if (choice == 3) accounts[i].deposit(amount); else accounts[i].withdraw(amount); } else if (choice == 5) { cout << "Enter receiver account: "; cin >> b; for (int k = 0; k < accounts.size(); k++) if (accounts[k].accNo == b) j = k; if (j == -1) cout << "Receiver Not Found!\n"; else { cout << "Enter amount: "; cin >> amount; accounts[i].transfer(accounts[j], amount); } } else if (choice == 6) accounts[i].display(); else accounts[i].showHistory(); } } while (choice != 8); cout << "Thank You!"; return 0; }
